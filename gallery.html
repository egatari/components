import csv
from statistics import mean


def read_rows(path):
    """CSV を読み、1 行を 1 つの辞書にする"""
    with open(path, encoding="utf-8") as f:
        return list(csv.DictReader(f))


def clean(rows):
    """空の値と数でない値を除く"""
    out = []
    for r in rows:
        try:
            r["score"] = float(r["score"])
        except ValueError:
            continue
        out.append(r)
    return out


def by_class(rows):
    """組ごとに点数を集める"""
    groups = {}
    for r in rows:
        groups.setdefault(r["class"], []).append(r["score"])
    return groups


def report(groups):
    """組ごとの平均を、高い順に書く"""
    pairs = [(c, mean(v)) for c, v in groups.items()]
    for c, m in sorted(pairs, key=lambda p: -p[1]):
        print(f"{c}: {m:.1f}")


rows = clean(read_rows("scores.csv"))
report(by_class(rows))
